/* ===================== PARTICLES CANVAS ===================== */
const canvas = document.getElementById('bg-canvas');
const ctx = canvas.getContext('2d');

let particles = [];
let mouse = { x: null, y: null };

function resizeCanvas() {
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;
}
resizeCanvas();
window.addEventListener('resize', () => { resizeCanvas(); initParticles(); });

function initParticles() {
  particles = [];
  const count = Math.floor((canvas.width * canvas.height) / 15000);
  for (let i = 0; i < count; i++) {
    particles.push({
      x: Math.random() * canvas.width,
      y: Math.random() * canvas.height,
      vx: (Math.random() - 0.5) * 0.3,
      vy: (Math.random() - 0.5) * 0.3,
      r: Math.random() * 1.5 + 0.3,
      opacity: Math.random() * 0.5 + 0.1,
      color: Math.random() > 0.5 ? '124,58,237' : '236,72,153'
    });
  }
}
initParticles();

function drawParticles() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  particles.forEach((p, i) => {
    p.x += p.vx;
    p.y += p.vy;
    if (p.x < 0 || p.x > canvas.width) p.vx *= -1;
    if (p.y < 0 || p.y > canvas.height) p.vy *= -1;

    // Mouse attraction
    if (mouse.x) {
      const dx = mouse.x - p.x, dy = mouse.y - p.y;
      const dist = Math.sqrt(dx*dx+dy*dy);
      if (dist < 120) {
        p.vx += dx * 0.00005;
        p.vy += dy * 0.00005;
      }
    }

    // Speed cap
    const speed = Math.sqrt(p.vx*p.vx+p.vy*p.vy);
    if (speed > 0.8) { p.vx *= 0.8/speed; p.vy *= 0.8/speed; }

    ctx.beginPath();
    ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
    ctx.fillStyle = `rgba(${p.color},${p.opacity})`;
    ctx.fill();

    // Connections
    for (let j = i + 1; j < particles.length; j++) {
      const q = particles[j];
      const dx = p.x - q.x, dy = p.y - q.y;
      const d = Math.sqrt(dx*dx+dy*dy);
      if (d < 100) {
        ctx.beginPath();
        ctx.moveTo(p.x, p.y);
        ctx.lineTo(q.x, q.y);
        ctx.strokeStyle = `rgba(124,58,237,${0.08 * (1 - d/100)})`;
        ctx.lineWidth = 0.5;
        ctx.stroke();
      }
    }
  });
  requestAnimationFrame(drawParticles);
}
drawParticles();

/* ===================== CONTACT FORM LOGIC ===================== */
document.addEventListener('DOMContentLoaded', function() {
  const FIELDS = ['cf-name','cf-email','cf-subject','cf-message'];
  const DRAFT_KEY = 'hassan-contact-draft';

  // Restore draft
  try {
    const draft = JSON.parse(localStorage.getItem(DRAFT_KEY) || '{}');
    let hasDraft = false;
    FIELDS.forEach(id => {
      const el = document.getElementById(id);
      if (el && draft[id]) { el.value = draft[id]; hasDraft = true; }
    });
    const cnt = document.getElementById('cf-message');
    if (cnt) cfMsgCount(cnt);
    if (hasDraft) { const n = document.getElementById('cf-draft-notice'); if(n) n.style.display='block'; }
  } catch(e) {}

  // Auto-save draft
  FIELDS.forEach(id => {
    const el = document.getElementById(id);
    if (!el) return;
    const saveEvt = function() {
      try {
        const data = {};
        FIELDS.forEach(f => { data[f] = document.getElementById(f)?.value || ''; });
        localStorage.setItem(DRAFT_KEY, JSON.stringify(data));
        const n = document.getElementById('cf-draft-notice'); if(n) n.style.display='block';
      } catch(e) {}
    };
    el.addEventListener('input', saveEvt);
    el.addEventListener('change', saveEvt);
  });

  // Submit
  const form = document.getElementById('contact-form');
  if (form) form.addEventListener('submit', function(e) { e.preventDefault(); if(validateCF()) submitCF(); });
});

function cfFocus(el) {
  el.style.borderColor = 'rgba(168,85,247,0.5)';
  el.style.background = 'rgba(124,58,237,0.06)';
  el.style.boxShadow = '0 0 0 3px rgba(124,58,237,0.1)';
}
function cfBlur(el) {
  if (!el.dataset.hasError) {
    el.style.borderColor = 'rgba(255,255,255,0.08)';
    el.style.background = 'rgba(255,255,255,0.03)';
  }
  el.style.boxShadow = 'none';
}
function cfMsgCount(el) {
  const c = document.getElementById('cf-msg-count');
  if (c) c.textContent = el.value.length + '/500';
  cfClear('cf-message-err', el);
}
function cfClear(errId, el) {
  const err = document.getElementById(errId);
  if (err) { err.style.display = 'none'; err.textContent = ''; }
  if (el) {
    delete el.dataset.hasError;
    el.style.borderColor = 'rgba(255,255,255,0.08)';
    el.style.background = 'rgba(255,255,255,0.03)';
    el.style.boxShadow = 'none';
  }
}
function cfError(fieldId, errId, msg) {
  const err = document.getElementById(errId);
  const field = document.getElementById(fieldId);
  if (err) { err.textContent = '⚠ ' + msg; err.style.display = 'block'; }
  if (field) {
    field.dataset.hasError = '1';
    field.style.borderColor = 'rgba(239,68,68,0.6)';
    field.style.background = 'rgba(239,68,68,0.04)';
    field.style.boxShadow = '0 0 0 3px rgba(239,68,68,0.08)';
    field.animate([
      {transform:'translateX(0)'},{transform:'translateX(-5px)'},{transform:'translateX(5px)'},
      {transform:'translateX(-3px)'},{transform:'translateX(3px)'},{transform:'translateX(0)'}
    ], {duration:320, easing:'ease-out'});
  }
}
function validateCF() {
  let ok = true;
  const name = (document.getElementById('cf-name')?.value || '').trim();
  const email = (document.getElementById('cf-email')?.value || '').trim();
  const subject = document.getElementById('cf-subject')?.value || '';
  const message = (document.getElementById('cf-message')?.value || '').trim();
  if (name.length < 2) { cfError('cf-name','cf-name-err','الرجاء إدخال الاسم (حرفين على الأقل)'); ok=false; }
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) { cfError('cf-email','cf-email-err','الرجاء إدخال بريد إلكتروني صحيح'); ok=false; }
  if (!subject) { cfError('cf-subject','cf-subject-err','الرجاء اختيار موضوع التواصل'); ok=false; }
  if (message.length < 10) { cfError('cf-message','cf-message-err','الرجاء كتابة رسالة لا تقل عن 10 أحرف'); ok=false; }
  return ok;
}
function submitCF() {
  const btn = document.getElementById('cf-submit');
  const icon = document.getElementById('cf-submit-icon');
  const text = document.getElementById('cf-submit-text');
  btn.disabled = true; btn.style.opacity = '0.75';
  icon.className = 'fas fa-spinner fa-spin text-sm';
  text.textContent = 'جاري الإرسال...';
  // Save submission history
  try {
    const submission = {
      name: document.getElementById('cf-name').value.trim(),
      email: document.getElementById('cf-email').value.trim(),
      subject: document.getElementById('cf-subject').value,
      message: document.getElementById('cf-message').value.trim(),
      timestamp: new Date().toISOString()
    };
    const hist = JSON.parse(localStorage.getItem('hassan-contact-history') || '[]');
    hist.unshift(submission);
    localStorage.setItem('hassan-contact-history', JSON.stringify(hist.slice(0,10)));
    localStorage.removeItem('hassan-contact-draft');
  } catch(e) {}
  setTimeout(function() {
    const successEl = document.getElementById('form-success');
    const successIcon = document.getElementById('success-icon');
    if (successEl) { successEl.style.opacity = '1'; successEl.style.pointerEvents = 'auto'; }
    setTimeout(function() { if(successIcon) successIcon.style.transform = 'scale(1)'; }, 80);
    btn.disabled = false; btn.style.opacity = '1';
    icon.className = 'fas fa-paper-plane text-sm';
    text.textContent = 'إرسال الرسالة';
  }, 1600);
}
function resetContactForm() {
  const successEl = document.getElementById('form-success');
  const successIcon = document.getElementById('success-icon');
  if (successEl) { successEl.style.opacity = '0'; successEl.style.pointerEvents = 'none'; }
  if (successIcon) successIcon.style.transform = 'scale(0)';
  ['cf-name','cf-email','cf-subject','cf-message'].forEach(id => {
    const el = document.getElementById(id);
    if (!el) return;
    el.value = '';
    delete el.dataset.hasError;
    el.style.borderColor = 'rgba(255,255,255,0.08)';
    el.style.background = 'rgba(255,255,255,0.03)';
    el.style.boxShadow = 'none';
  });
  const c = document.getElementById('cf-msg-count'); if(c) c.textContent='0/500';
  const n = document.getElementById('cf-draft-notice'); if(n) n.style.display='none';
  ['cf-name-err','cf-email-err','cf-subject-err','cf-message-err'].forEach(id => {
    const e = document.getElementById(id); if(e){e.style.display='none';e.textContent='';}
  });
}

/* ===================== THEME TOGGLE ===================== */
(function() {
  const html = document.documentElement;
  const btn = document.getElementById('theme-toggle');
  const knob = document.getElementById('theme-knob');

  // Load saved theme
  const saved = localStorage.getItem('hassan-theme') || 'dark';
  applyTheme(saved);

  btn.addEventListener('click', () => {
    const current = html.getAttribute('data-theme') || 'dark';
    const next = current === 'dark' ? 'light' : 'dark';
    applyTheme(next);
    localStorage.setItem('hassan-theme', next);
  });

  function applyTheme(theme) {
    html.setAttribute('data-theme', theme);
    if (theme === 'light') {
      knob.textContent = '☀️';
      document.body.style.background = '#f4f4f9';
    } else {
      knob.textContent = '🌙';
      document.body.style.background = '#030305';
    }
  }
})();

/* ===================== SCROLL TO TOP LOGIC ===================== */
(function() {
  const btn = document.getElementById('scroll-to-top');
  if (!btn) return;

  window.addEventListener('scroll', function() {
    if (window.scrollY > 400) {
      btn.classList.add('visible');
    } else {
      btn.classList.remove('visible');
    }
  }, { passive: true });

  btn.addEventListener('click', function() {
    window.scrollTo({ top: 0, behavior: 'smooth' });
  });
})();

(function() {
  const catCounts = { react: 0, laravel: 0, ai: 0, flutter: 0, cli: 0 };
  const cards = document.querySelectorAll('.project-card[data-category]');
  let totalAll = cards.length;

  // Count categories
  cards.forEach(card => {
    const cats = (card.getAttribute('data-category') || '').trim().split(/\s+/).filter(Boolean);
    cats.forEach(c => { if (catCounts[c] !== undefined) catCounts[c]++; });
  });

  // Update counts in buttons
  const el = id => document.getElementById(id);
  if (el('cnt-all')) el('cnt-all').textContent = totalAll;
  Object.keys(catCounts).forEach(k => { if (el('cnt-' + k)) el('cnt-' + k).textContent = catCounts[k]; });

  // Filter logic
  function filterCards(category) {
    let delay = 0;
    cards.forEach(card => {
      const cats = (card.getAttribute('data-category') || '').trim().split(/\s+/).filter(Boolean);
      const show = category === 'all' || cats.includes(category);

      if (show) {
        // Show: remove hidden, add visible with staggered delay
        card.style.transitionDelay = delay * 60 + 'ms';
        card.classList.remove('filter-hidden');
        card.classList.add('filter-visible');
        delay++;
      } else {
        card.style.transitionDelay = '0ms';
        card.classList.remove('filter-visible');
        card.classList.add('filter-hidden');
      }
    });
  }

  // Bind filter buttons
  const filterBtns = document.querySelectorAll('.filter-btn[data-filter]');
  filterBtns.forEach(btn => {
    btn.addEventListener('click', () => {
      filterBtns.forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      // Micro-bounce on active button
      btn.animate([
        {transform:'scale(1)'},{transform:'scale(0.93)'},{transform:'scale(1.05)'},{transform:'scale(1)'}
      ], {duration:300, easing:'cubic-bezier(0.34,1.56,0.64,1)'});
      filterCards(btn.getAttribute('data-filter'));
    });
  });
})();

/* ===================== AOS INIT ===================== */
AOS.init({ duration: 800, easing: 'ease-out-cubic', once: true, offset: 60 });
